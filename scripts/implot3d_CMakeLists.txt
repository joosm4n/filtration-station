cmake_minimum_required(VERSION 3.16)
project(implot3d CXX)

# Drop this file into the root of a manual implot3d clone (next to implot3d.cpp).
# Needs the `implot3d` target to exist first (add_subdirectory(external/implot3d) before this one).

add_library(implot3d STATIC
  ${CMAKE_CURRENT_SOURCE_DIR}/implot3d.cpp
  ${CMAKE_CURRENT_SOURCE_DIR}/implot3d_items.cpp
  ${CMAKE_CURRENT_SOURCE_DIR}/implot3d_demo.cpp
  ${CMAKE_CURRENT_SOURCE_DIR}/implot3d_meshes.cpp
)
target_include_directories(implot3d PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(implot3d PUBLIC cxx_std_17)
target_link_libraries(implot3d PUBLIC imgui)
